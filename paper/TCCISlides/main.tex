    \begin{columns}[t]
    \begin{column}{0.45\textwidth}

            \end{itemize}
    \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Problema e Justificativa}
    \begin{itemize}
        \item Sistemas de distribuição de conteúdo $\approx$ múltiplos servidores, um balanceador de carga;
        \item O uso de memória (principal e secundária) é correlacionado com o desempenho de servidores de distribuição de conteúdo e é um cenário comum em CDNs (\textit{Content Delivery Networks}) com \textbf{servidores de borda} (\textit{Edge Servers} ou \textit{Cache Servers}) \cite{surveyCDNArchitectures2025};
        \item Há uma lacuna em estratégias convencionais de balanceamento de carga que ignoram o estado dos recursos de memória ou consideram outros recursos computacionais em conjunto;
    \end{itemize}

    \vfill
    {\scriptsize \textit{\textbf{Observação}: Os termos "uso de memória" e "consumo de memória", no contexto deste trabalho, correspondem às métricas de \textbf{consumo de memória principal} e \textbf{taxa de entrada e saída (I/O) de memória secundária}.}}
\end{frame}

\begin{frame}{Objetivo}
    \textbf{Objetivo do trabalho}
    \begin{itemize}
        \item O objetivo do trabalho é seguir pela abordagem da consideração exclusiva do uso de memória de modo a avaliar uma estratégia de otimização da escolha do servidor em balanceadores de carga para sistemas de distribuição de conteúdo.
        \item O objetivo será atingido por meio da implementação, amparada pelo estabelecimento de uma arquitetura de simulação para testes, de um algoritmo de balanceamento de carga e posterior análise comparativa.
    \end{itemize}
\end{frame}

\section{Trabalhos relacionados}

\begin{frame}[allowframebreaks]{Trabalhos Relacionados}

\begin{itemize}
    \item \textbf{NGINX:} Sistema popular que utiliza algoritmos como \textit{Round-Robin} que podem levar a desequilíbrios, pois não monitoram o \textbf{uso de memória} dos servidores em tempo real para tomar decisões \cite{nginxAlgorithmsEvaluationYuhong2022}.

    \item \textbf{HAProxy:} É reconhecido pela alta eficiência e baixo consumo de recursos, mas também carece de um mecanismo nativo de monitoramento de memória para otimizar a distribuição de tráfego \cite{haproxy}.

    \item \textbf{Docker Swarm:} Embora um estudo tenha explorado adicionar a \textbf{influência da memória} em seu balanceador, o foco foi exclusivo do ecossistema Docker, sem aplicação em servidores de distribuição de conteúdo \cite{dockerswarmmemory2018}.

    \item \textbf{Algoritmo Dinâmico (Cloud):} Uma solução foi proposta para lidar com \textit{Flash Crowds} em aplicações na nuvem usando múltiplas métricas, mas sem especificidade para o gargalo de memória em CDNs \cite{novelWeightAssignmentLoadBalancingAlgorithmForCloudApps2023}.
    
    \item \textbf{Revisão de Parâmetros:} Uma revisão de algoritmos dinâmicos mostrou que, embora alguns usem \textbf{memória e disco} como parâmetros, nenhum os utiliza como fator principal ou exclusivo de avaliação de desempenho \cite{webServerPerformanceImprovement2021Review}.
    
    \item \textbf{Streaming de Vídeo (iSCON):} A estratégia iSCON foca em replicar conteúdo popular para balancear a carga, mas utiliza a \textbf{largura de banda} como principal indicador, ignorando a pressão sobre a memória \cite{dynamicLoadBalancingForEfficientVideoStreamingService2015}.
    
    \item \textbf{Servidores de Vídeo Distribuídos:} Um trabalho propôs uma solução de duas fases que mede a carga pelo \textbf{número de streams ativas}, uma abstração que não considera gargalos de recursos de baixo nível como a memória \cite{dataAllocationAndDynamicLoadBalancingForDistributedVideoStorageServer1999}.
\end{itemize}

\end{frame}

\section{Fundamentação Teórica}

\begin{frame}{Content Delivery Networks (CDNs)}

\textbf{CDNs} são infraestruturas cruciais para entregar conteúdo rápido, confiável e eficiente, distribuindo-o em servidores geograficamente dispersos.

\begin{itemize}
    \item São \textbf{sistemas distribuídos de computadores}, administrados no nível da aplicação, que encaminham requisições para o servidor mais próximo do usuário, reduzindo latência e sobrecarga \cite{Preston2021Finding};
    \item \textbf{Arquitetura}: Servidores de origem (conteúdo original) e servidores de borda (cópias redundantes) em PoPs (\textit{Points of Presence});
    \item \textbf{Cache}: Conteúdo é servido do PoP mais próximo, otimizando a distância física e reduzindo a carga na rede global;
    \item \textbf{Roteamento}: Utilizam DNS para direcionar usuários ao servidor ideal com base na localização geográfica;
    \item \textbf{Balanceamento de Carga}: Distribuem requisições para evitar sobrecarga em PoPs, usando localização geográfica e técnicas como \textit{anycast} \cite{Preston2021Finding};
\end{itemize}
\end{frame}

% \subsection{Balanceadores de carga}

\begin{frame}{Balanceadores de Carga}

Balanceadores de carga são cruciais em sistemas distribuídos, incluindo CDNs.

\textbf{Função e Importância}
\begin{itemize}
    \item Distribuem eficientemente cargas de trabalho (requisições, tráfego) entre múltiplos servidores e são essenciais para a \textbf{escalabilidade horizontal}, preferencial à vertical por flexibilidade e custo \cite{scalabilityIssuesInDatabaseSystems};
    \item Melhoram o desempenho, a escalabilidade, a confiabilidade e a disponibilidade (ex: redirecionamento em caso de falha);
    \item A eficácia do sistema distribuído depende da sofisticação do balanceamento de carga;
    \item Categorias principais: \textbf{Nível de aplicação} e \textbf{nível de rede}; \cite{Pandian2022ICCBI};
\end{itemize}
\end{frame}

% \subsection{MPEG-DASH}

\begin{frame}{MPEG-DASH}

\textbf{\textit{HTTP (Hypertext Transfer Protocol) Streaming}} é padrão para multimídia, levando ao \textbf{DASH} (\textit{Dynamic Adaptive Streaming over HTTP}) \cite{thang2012adaptive}.

\textbf{DASH: Arquitetura e Componentes}
\begin{itemize}
    \item \textbf{Vantagem}: Abordagem \textit{stateless} do HTTP é escalável, contrastando com RTP \cite{sodagar2011mpegdash};
    \item \textbf{Orientado ao Cliente}: Servidor disponibiliza; cliente gerencia a inteligência da sessão \cite{stockhammer2011dynamic};
    \item \textbf{MPD (Media Presentation Description)}: Manifesto XML com URLs e versões do conteúdo (Períodos, Adaptation Sets, Representações) \cite{sodagar2011mpegdash};
    \item \textbf{Segmentos}: Conteúdo dividido em blocos, requisitados via HTTP GET, facilmente \textit{cacheáveis} por CDNs \cite{stockhammer2011dynamic};
    \item \textbf{Content Steering}: Cliente decide \textit{de onde} buscar o conteúdo via múltiplas URLs base no MPD \cite{sodagar2011mpegdash};
\end{itemize}
\end{frame}

\section{Proposta}

\subsection{Algoritmo}

\begin{frame}{Objetivo do Algoritmo}

\begin{itemize}
    \item O objetivo principal é definir o método de escolha de um servidor para cada nova requisição;
    
    \bigskip
    
    \item A natureza do algoritmo é \textbf{probabilística};
    \begin{itemize}
        \item A chance de um servidor ser escolhido é baseada em uma probabilidade calculada dinamicamente (de acordo com a Equação \mbox{\ref{eq:basic_li}});
    \end{itemize}
    
    \bigskip
    
    \item A probabilidade é influenciada principalmente pelo Consumo de \textbf{Memória RAM} e pela taxa de entrada e saída de \textbf{Disco (I/O)};
    
    \bigskip
    
    \item O funcionamento do algoritmo é \textbf{dinâmico};
    \begin{itemize}
        \item Os servidores enviam dados de monitoramento (carga) periodicamente para o balanceador, que atualiza os parâmetros para as próximas requisições;
    \end{itemize}
\end{itemize}

\end{frame}

\begin{frame}{Cálculo da probabilidade}

A probabilidade ($P_i$) de uma requisição ser enviada para um servidor $i$ é baseada em uma versão adaptada do algoritmo \textit{Basic Load Interpretation} (\textit{Basic LI}) \cite{interpretingStaleLoadInformation2000}. A equação utilizada corresponde exatamente à equação determinada pelo \textit{Basic LI}:

\begin{equation}\label{eq:basic_li}
    P_{i}=\frac{\frac{L_{tot}+arrive_{T}}{n}-L_{i}}{arrive_{T}}
\end{equation}

\begin{center}
\begin{minipage}{0.9\textwidth}
\begin{itemize}
    \item[$P_i$]: Probabilidade de escolha do servidor $i$;
    \item[$n$]: Número total de servidores disponíveis;
    \item[$L_i$]: Carga individual reportada pelo servidor $i$;
    \item[$L_{tot}$]: Somatório de todas as cargas reportadas ($\sum L_i$);
    \item[$arrive_T$]: Número total de requisições esperadas no sistema inteiro para o tempo médio ($T$) das informações de carga ($\sum R_i$);
\end{itemize}
\end{minipage}
\end{center}

\end{frame}

\begin{frame}{Adaptação do Parâmetro de Carga ($L_i$)}

No contexto deste trabalho, o termo "carga" ($L_i$) é adaptado (de tamanho de fila de requisições para uso de memória) para refletir o desempenho de um servidor de conteúdo. A carga é calculada a partir de métricas específicas por meio da Equação \ref{eq:limemory}.

\begin{equation}\label{eq:limemory}
    L_i = CM \cdot LM_i + CD \cdot LD_i
\end{equation}

\begin{center}
\begin{minipage}{0.8\textwidth}
\begin{itemize}
    \item[$LM_i$]: Consumo de \textbf{memória} normalizado do servidor $i$ (em \%);
    \bigskip
    \item[$LD_i$]: Carga de \textbf{disco} (I/O) normalizada do servidor $i$ (em \%);
    \bigskip
    \item[$CM, CD$]: Coeficientes dinâmicos que representam a importância geral da memória e do disco no sistema, respectivamente;
\end{itemize}
\end{minipage}
\end{center}

\end{frame}

\begin{frame}{Cálculo dos Coeficientes Dinâmicos ($CM$ e $CD$)}

Os coeficientes dão mais peso ao recurso que está mais sobrecarregado no sistema como um todo. Eles são calculados a partir das taxas de uso gerais de memória ($TM$) e disco ($TD$). A propriedade $CM + CD = 1$ garante que a importância seja distribuída de forma proporcional entre os dois recursos.

\bigskip

\begin{itemize}
    \item Taxa de Memória: $TM = \sum_{i=0}^n \frac{LM_i}{n}$
    \bigskip
    \item Taxa de Disco: $TD = \sum_{i=0}^n \frac{LD_i}{n}$
\end{itemize}

\bigskip

\noindent
\begin{minipage}{.48\textwidth}
    \begin{equation}
        CM = \frac{TM}{TM + TD} \label{eq:cm}
    \end{equation}
\end{minipage}
\hfill
\begin{minipage}{.48\textwidth}
    \begin{equation}
        CD = \frac{TD}{TM + TD} \label{eq:cd}
    \end{equation}
\end{minipage}

O algoritmo é executado a cada nova requisição que chega ao balanceador e sua execução é determinada conforme o pseudo-código a seguir.

\end{frame}

\begin{frame}{Pseudocódigo}
    \begin{algorithm}[H]
    \caption{Seleção de Servidor por Requisição}
    \begin{algorithmic}[1]
    \Statex \textbf{Gatilho}: Chegada de uma nova requisição $R$.
    \Require
        \Statex $S$: Lista de servidores com métricas $L_i$, $R_i$ e $T_i$ atualizadas.
        \Statex Variáveis globais: $L_{tot}, n$.
    \Ensure
        \Statex Requisição $R$ encaminhada para um servidor $S_{i}$.
    
    \Procedure{SelecionarServidor}{$R, S, L_{tot}, n$}
        \State $arrive_T \leftarrow \text{CalcularQtdRequisicoesEsperadas}(S)$
        \State $P_{list} \leftarrow []$ \Comment{Lista de probabilidades}
        
        \ForAll{$S_i, L_i$ in $S$}
            \State $P_i \leftarrow ((L_{tot} + arrive_T)/n - L_i) / arrive_T$ \Comment{Eq. \ref{eq:basic_li}}
            \State Adicionar $(S_i, P_i)$ a $P_{list}$
        \EndFor
        
        \State $s_{escolhido} \leftarrow \text{SelecaoAleatoriaComPesos}(P_{list})$
        \State \text{Encaminhar}(R, $s_{escolhido}$)
    \EndProcedure
    \end{algorithmic}
    \end{algorithm}
\end{frame}

\begin{frame}{Metodologia de Validação e Métricas}

\begin{itemize}
    \item \textbf{Estratégia de Validação}:
    \begin{itemize}
        \item Análise comparativa rigorosa executada em cenários de \textbf{alta carga};

    \bigskip % Espaço vertical para clareza
    
    \item \textbf{Definição de Sobrecarga}:
    \begin{itemize}
        \item Ponto definido empiricamente quando a Experiência do Usuário (QoE) se degrada, com \textbf{travamentos visíveis} (\textit{stalls}) no player de vídeo;
    \end{itemize}
    
    \bigskip
    
    \item \textbf{Métricas de Avaliação (QoE)}:
    \begin{itemize}
        \item Número total de travamentos (\textit{stalls}) do vídeo;
        \item Duração média de cada travamento;
        \item Tempo total de ociosidade do serviço (soma das pausas);
    \end{itemize}
\end{itemize}

\end{frame}

\section{Considerações parciais}

\begin{frame}{Proposta Central do Trabalho}

\begin{itemize}
    \item \textbf{O Problema Identificado}:
    \begin{itemize}
        \item Estratégias clássicas (\textit{Round-Robin}, Aleatória) ignoram o estado real dos recursos dos servidores, impactando a Qualidade de Experiência (QoE);
        \item A \textbf{memória} (principal e secundária) é um fator de gargalo crítico em servidores que entregam conteúdo a partir de arquivos;
    \end{itemize}
    
    \item \textbf{A Solução Proposta}:
    \begin{itemize}
        \item Um algoritmo de balanceamento \textbf{probabilístico} que integra o monitoramento de recursos como fator decisório;
        \item Adapta o conceito de \textit{Load Interpretation} para considerar não apenas o consumo de recursos, mas também a \textbf{idade da informação};
    \end{itemize}
\end{itemize}

\end{frame}
